// This oracle only accepts the already hand-audited scalar/integer fixtures.
// It deliberately does not use the implementation's canonicalizer or edge walk.
function literalCanonical(value: any): string {
  if (Array.isArray(value)) return '[' + value.map(literalCanonical).join(',') + ']';
  if (value !== null && typeof value === 'object') {
    return '{' + Object.keys(value).sort().map(k => JSON.stringify(k) + ':' + literalCanonical(value[k])).join(',') + '}';
  }
  return JSON.stringify(value);
}
function literalIdentity(row: Row): string {
  const payload = Object.fromEntries(Object.entries(row).filter(([key]) => key !== 'id'));
  return 'sha256:' + createHash('sha256').update('pk/v1/' + row.kind + '\n' + literalCanonical(payload), 'utf8').digest('hex');
}
function tables(audit: Row): Record<string, Row[]> {
  return Object.fromEntries(Object.entries(audit.tables).map(([kind, rows]) => [kind,
    (Array.isArray(rows) ? rows : Object.values(rows as Row)).slice().sort((a: Row, b: Row) => a.id.localeCompare(b.id)),
  ]));
}

// Explicit positions independently inspected against the normative join prose.
// Legacy IDs and arbitrary text never enter this list or get rewritten.
const singleEdges: Record<string, string[]> = {
  origin: [], source: ['origin'], prefix: ['origin'], evidence: ['origin', 'source', 'prefix'],
  certificate: ['source'], fact: ['certificate'], starting: ['origin', 'source', 'parent'],
  current: ['origin', 'prefix', 'parent'], request: ['origin', 'source', 'prefix'],
  knowledge: ['origin', 'starting', 'current', 'request'], profile: [],
  verdict: ['knowledge', 'profile'], boundary: ['origin', 'parent', 'restored_from'],
  binding: ['origin', 'input_knowledge', 'successor_knowledge'],
};
const listEdges: Record<string, string[]> = {
  certificate: ['witnesses'], fact: ['evidence'], starting: ['facts'], current: ['facts'], request: ['facts'],
};
function reseal(audit: Row): Row {
  const expanded = tables(audit);
  const byOldId = new Map<string, Row>();
  for (const rows of Object.values(expanded)) for (const row of rows) byOldId.set(row.id, row);
  const completed = new Map<string, string>();
  const active = new Set<string>();
  function replace(id: string | null): string | null {
    if (id === null || !byOldId.has(id)) return id;
    if (completed.has(id)) return completed.get(id)!;
    assert.ok(!active.has(id), 'test mutation introduced a cycle');
    active.add(id);
    const row = byOldId.get(id)!;
    for (const field of singleEdges[row.kind]) row[field] = replace(row[field]);
    for (const field of listEdges[row.kind] ?? []) row[field] = row[field].map(replace).sort();
    if (row.kind === 'starting') for (const crosswalk of row.crosswalk) crosswalk.certificate = replace(crosswalk.certificate);
    if (row.kind === 'boundary') for (const binding of row.bindings) {
      binding.knowledge = replace(binding.knowledge); binding.verdict = replace(binding.verdict);
    }
    row.id = literalIdentity(row);
    active.delete(id); completed.set(id, row.id);
    return row.id;
  }
  for (const id of byOldId.keys()) replace(id);
  for (const field of ['origin', 'required_profile', 'target_boundary']) audit[field] = replace(audit[field]);
  audit.tables = Object.fromEntries(Object.entries(expanded).map(([kind, rows]) => [kind,
    audit.form === 'compact'
      ? Object.fromEntries(rows.sort((a, b) => a.id.localeCompare(b.id)).map(row => [row.id, row]))
      : rows.sort((a, b) => a.id.localeCompare(b.id)),
  ]));
  audit.id = literalIdentity(audit);
  return audit;
}
function assertSealed(audit: Row): void {
  assert.equal(audit.id, literalIdentity(audit));
  for (const rows of Object.values(tables(audit))) for (const row of rows) assert.equal(row.id, literalIdentity(row));
}
function rejected(value: any): void {
  const before = literalCanonical(value);
  assert.throws(() => checkKnowledgeStructure(value), Error);
  assert.equal(literalCanonical(value), before, 'failed validation mutated input');
  assert.throws(() => serializeKnowledgeWire(value), Error);
  assert.equal(literalCanonical(value), before, 'failed serialization mutated input');
}

test('16 independent literal preimages cover every identity domain and Unicode', () => {
  const vectors = load('identity-vectors.json');
  assert.equal(vectors.length, 16);
  assert.equal(new Set(vectors.map((v: Row) => v.kind)).size, 16);
  for (const vector of vectors) {
    const preimage = Buffer.from(vector.preimage_utf8_hex, 'hex');
    assert.equal('sha256:' + createHash('sha256').update(preimage).digest('hex'), vector.id);
    const payload = Object.fromEntries(Object.entries(vector.object).filter(([key]) => key !== 'id'));
    assert.deepEqual(Buffer.concat([Buffer.from('pk/v1/' + vector.kind + '\n'), canonicalKnowledgeJson(payload)]), preimage);
    assert.equal(knowledgeIdentity(vector.object), vector.id);
    assert.equal(knowledgeIdentity({ ...vector.object, id: missing }), vector.id, 'only top-level id is excluded');
  }
});

test('raw ingress rejects lexical ambiguity before JSON.parse can erase it', () => {
  const invalid = [
    '{"a":1,"a":2}', '{"a":1,"\\u0061":2}', '{"nest":{"a":1,"a":1}}',
    '-0', '1.0', '1e0', '1E+0', '[0,-0]', '{"a":1.25}', 'NaN', 'Infinity',
    '{"s":"\\ud800"}', '{"s":"\\udfff"}', '{"\\ud800":0}', '"\ud800"',
    '\ufeff{}', '{}{}', '/*comment*/{}', '{"a":1,}', 'true false',
  ];
  for (const text of invalid) assert.throws(() => parseKnowledgeJson(text), Error, text);
  for (const bytes of [Buffer.from([0xff]), Buffer.from([0xc0, 0xaf]), Buffer.from([0xed, 0xa0, 0x80]), Buffer.from([0xf0, 0x9f]), Buffer.from([0xef, 0xbb, 0xbf, 0x7b, 0x7d])]) {
    assert.throws(() => parseKnowledgeJson(bytes), Error, bytes.toString('hex'));
  }
  assert.deepEqual(parseKnowledgeJson(' {"zero":0,"integer":2147483647,"flag":true} \r\n'), { zero: 0, integer: 2147483647, flag: true });
  assert.equal(parseKnowledgeJson('1'), 1, 'object integer 1 cannot preserve whether callers once wrote 1.0');
  assert.throws(() => parseKnowledgeJson('['.repeat(33) + '0' + ']'.repeat(33)), Error);
  assert.throws(() => parseKnowledgeJson(' '.repeat(16 * 1024 * 1024) + '{}'), Error);
});

test('same-source refinement retains proved facts/crosswalk, changed-source roots reprove, suffix is causal', () => {
  const old = tables(load('replay-full.json')); const suffix = tables(load('replay-suffix-full.json'));
  const child = old.starting.find(row => row.parent !== null)!;
  const parent = old.starting.find(row => row.id === child.parent)!;
  assert.equal(child.revision, parent.revision + 1); assert.equal(child.source, parent.source);
  const facts = new Map(old.fact.map(row => [row.id, row]));
  const proved = parent.facts.filter((id: string) => facts.get(id)!.state !== 'unknown');
  assert.ok(proved.length > 0); assert.ok(proved.every((id: string) => child.facts.includes(id)));
  assert.ok(parent.facts.some((id: string) => facts.get(id)!.state === 'unknown'));
  assert.ok(child.facts.every((id: string) => facts.get(id)!.state === 'known'));
  assert.ok(parent.crosswalk.every((row: Row) => child.crosswalk.some((candidate: Row) => literalCanonical(row) === literalCanonical(candidate))));
  assert.ok(suffix.starting.every(row => row.revision === 0 && row.parent === null));
  assert.ok(suffix.starting.every(row => !old.starting.some(prior => prior.source === row.source)));

    const current = t.current.filter(row => row.cursor <= 5);

    const fact = t.fact.filter(row => factIds.has(row.id));
    const evidenceIds = new Set(fact.flatMap(row => row.evidence));
    return { current, fact, evidence: t.evidence.filter(row => evidenceIds.has(row.id)), prefix: t.prefix.filter(row => row.records.length <= 5) };
  }
  for (const form of ['full', 'ordinary', 'compact']) assert.deepEqual(chronological(tables(load('replay-' + form + '.json'))), chronological(tables(load('replay-suffix-' + form + '.json'))));

function revisionCrosswalk(owner: string, form: string): Row {
  const audit = load('replay-full.json');
  if (owner === 'p1') {
    const start = audit.tables.starting.find((s: Row) => s.owner === owner);
    const cert = structuredClone(audit.tables.certificate.find((c: Row) => c.owner === owner));
    cert.property = 'identity_reveal'; cert.id = literalIdentity(cert);
    audit.tables.certificate.push(cert);
    start.crosswalk = [{appearance:'p1:appearance:1', roster:'p1:roster:1', certificate:cert.id}];
  }
  audit.form = form; return reseal(audit);
}
function revisionNested(count: number, style: string, leaf: any): any {
  let value: any = style === 'object' ? {} : [];
  if (leaf !== 'empty') value = style === 'object' ? {x:leaf} : [leaf];
  for (let index = 1; index < count; index++) value = style === 'object' || style === 'mixed' && index % 2 ? {x:value} : [value];
  return value;
}
function revisionCheck(audit: Row, guard?: RegExp): void {
  assertSealed(audit); const before = literalCanonical(audit);
  if (guard) {
    assert.throws(() => checkKnowledgeStructure(audit), guard);
    assert.throws(() => serializeKnowledgeWire(audit), guard);
  } else {
    assert.deepEqual(checkKnowledgeStructure(audit), diagnostic);
    assert.deepEqual(checkKnowledgeStructure(parseKnowledgeJson(serializeKnowledgeWire(audit))), diagnostic);
  }
  assert.equal(literalCanonical(audit), before);
}

test('revision: mirrored owner crosswalks preserve exact certificate joins in all forms', () => {
  for (const owner of ['p1','p2']) {
    for (const form of ['full','ordinary','compact']) {
      revisionCheck(revisionCrosswalk(owner,form));
      const audit = revisionCrosswalk(owner,form), expanded = tables(audit);
      const start = expanded.starting.find(s => s.owner === owner && s.crosswalk.length)!;
      const cross = start.crosswalk[0], cert = expanded.certificate.find(c => c.id === cross.certificate)!;
      cross.roster = cert.subject = (owner === 'p1' ? 'p2' : 'p1') + ':roster:1';
      revisionCheck(reseal(audit), /crosswalk owner/);
    }
    for (const field of ['appearance','subject','owner','source']) {
      const audit = revisionCrosswalk(owner,'full');
      const start = audit.tables.starting.find((s: Row) => s.owner === owner && s.crosswalk.length);
      const cross = start.crosswalk[0], cert = audit.tables.certificate.find((c: Row) => c.id === cross.certificate);
      if (field === 'appearance') cross.appearance = (owner === 'p1' ? 'p2' : 'p1') + ':appearance:1';
      else if (field === 'subject') cert.subject = owner + ':roster:2';
      else if (field === 'owner') cert.owner = owner === 'p1' ? 'p2' : 'p1';
      else {
        const source = structuredClone(audit.tables.source[0]); source.records[0] = '|turn|1'; source.id = literalIdentity(source);
        audit.tables.source.push(source); cert.source = source.id;
        cert.witnesses = cert.witnesses.map((id: string) => {
          const witness = structuredClone(audit.tables.evidence.find((e: Row) => e.id === id));
          witness.source = source.id; witness.id = literalIdentity(witness); audit.tables.evidence.push(witness); return witness.id;
        });
      }
      revisionCheck(reseal(audit), /crosswalk owner|certificate witness|crosswalk certificate source/);
    }
  }
});
